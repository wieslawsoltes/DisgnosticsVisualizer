cmake_minimum_required(VERSION 3.20)
project(DiagnosticsVisualizerNative LANGUAGES CXX)
set(RUNTIME_SDK_ROOT "" CACHE PATH "Pinned dotnet/runtime SDK sources")
if(NOT EXISTS "${RUNTIME_SDK_ROOT}/src/coreclr/pal/prebuilt/inc/corprof.h")
  message(FATAL_ERROR "Supply -DRUNTIME_SDK_ROOT with the pinned profiling SDK; see native/README.md")
endif()
set(SDK_TEST "${RUNTIME_SDK_ROOT}/src/tests/profiler/native")
set(SDK_CLR "${RUNTIME_SDK_ROOT}/src/coreclr")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(REWRITER_DIR "${CMAKE_CURRENT_BINARY_DIR}/runtime-rewriter")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare_sdk.py" "${RUNTIME_SDK_ROOT}" "${REWRITER_DIR}" COMMAND_ERROR_IS_FATAL ANY)
add_library(DiagnosticsVisualizerProfiler SHARED profiler.cpp capture.cpp pal_compat.cpp
  "${SDK_TEST}/profiler.cpp" "${SDK_TEST}/guids.cpp"
  "${REWRITER_DIR}/ilrewriter.cpp" "${REWRITER_DIR}/sigparse.cpp")
target_compile_features(DiagnosticsVisualizerProfiler PRIVATE cxx_std_20)
target_compile_definitions(DiagnosticsVisualizerProfiler PRIVATE SOS_INCLUDE HOST_64BIT NOMINMAX)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  target_compile_definitions(DiagnosticsVisualizerProfiler PRIVATE HOST_AMD64 TARGET_AMD64)
elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64|ARM64)$")
  target_compile_definitions(DiagnosticsVisualizerProfiler PRIVATE HOST_ARM64 TARGET_ARM64)
else()
  message(FATAL_ERROR "Only x64 and ARM64 hosts are supported")
endif()
target_include_directories(DiagnosticsVisualizerProfiler PRIVATE "${REWRITER_DIR}" "${SDK_TEST}" "${SDK_TEST}/rejitprofiler"
  "${SDK_CLR}/pal/prebuilt/inc" "${SDK_CLR}/inc" "${RUNTIME_SDK_ROOT}/src/native")
if(WIN32)
  target_link_libraries(DiagnosticsVisualizerProfiler PRIVATE Advapi32 oleaut32 Ole32)
  set_target_properties(DiagnosticsVisualizerProfiler PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS OFF)
else()
  target_include_directories(DiagnosticsVisualizerProfiler PRIVATE "${SDK_CLR}/pal/inc/rt" "${SDK_CLR}/pal/inc")
  target_compile_options(DiagnosticsVisualizerProfiler PRIVATE -fvisibility=hidden -fno-strict-aliasing)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(DiagnosticsVisualizerProfiler PRIVATE -Wno-null-arithmetic)
  else()
    target_compile_options(DiagnosticsVisualizerProfiler PRIVATE -Wno-conversion-null -Wno-pointer-arith)
  endif()
  target_link_libraries(DiagnosticsVisualizerProfiler PRIVATE pthread ${CMAKE_DL_LIBS})
endif()
install(TARGETS DiagnosticsVisualizerProfiler DESTINATION native)
